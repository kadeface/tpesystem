## 3. 工作流程设计

### 3.1 特征生成流程

```
选择考试 -> 区级特征计算 -> 校级特征计算 -> 班级特征计算 -> 学生特征计算 -> 返回结果
```

特征计算采用自上而下的计算顺序，确保每一层次的特征计算都能依赖于已经计算完成的上层特征。

### 3.2 详细实现步骤

1. **区级特征计算**
   ```python
   def _generate_district_features(self, exam):
       # 删除已存在的特征
       DistrictExamFeature.objects.filter(exam=exam).delete()
       
       # 获取考试所有成绩记录
       scores = Score.objects.filter(exam=exam)
       
       # 计算基础统计指标
       stats = scores.aggregate(
           avg=Avg('raw_score'),
           std=StdDev('raw_score'),
           min=Min('raw_score'),
           max=Max('raw_score'),
           count=Count('id')
       )
       
       # 计算各学科平均分
       subject_stats = scores.values('subject').annotate(
           subject_id=F('subject'),
           avg=Avg('raw_score')
       )
       
       subject_avgs = {stat['subject_id']: stat['avg'] for stat in subject_stats}
       
       # 创建区级特征
       district_feature = DistrictExamFeature.objects.create(
           exam=exam,
           avg_score=stats['avg'],
           std_dev=stats['std'] or 0,
           total_students=stats['count'],
           max_score=stats['max'],
           min_score=stats['min'],
           subject_avg_scores=subject_avgs
       )
       
       return district_feature
   ```

2. **校级特征计算**
   ```python
   def _generate_school_features(self, exam, district_feature):
       # 删除已存在的特征
       SchoolExamFeature.objects.filter(exam=exam).delete()
       
       # 获取考试所有成绩记录
       scores = Score.objects.filter(exam=exam)
       
       # 按学校分组统计
       school_stats = scores.values('student__current_school').annotate(
           school_id=F('student__current_school'),
           avg_score=Avg('raw_score'),
           std_dev=StdDev('raw_score'),
           class_count=Count('student__current_class', distinct=True),
           student_count=Count('student', distinct=True),
           top_score=Max('raw_score')
       ).order_by('-avg_score')
       
       # 计算学校排名并创建特征
       school_features = {}
       for rank, school_stat in enumerate(school_stats, 1):
           school_id = school_stat['school_id']
           
           school_feature = SchoolExamFeature.objects.create(
               exam=exam,
               district_feature=district_feature,
               school_id=school_id,
               avg_score=school_stat['avg_score'],
               std_dev=school_stat['std_dev'] or 0,
               district_rank=rank,
               district_percentile=((len(school_stats) - rank) / len(school_stats)) * 100,
               class_count=school_stat['class_count'],
               top_student_score=school_stat['top_score']
           )
           
           school_features[school_id] = school_feature
       
       return school_features
   ```

3. **班级特征计算**
   ```python
   def _generate_class_features(self, exam, scores, school_features):
       # 删除已存在的特征
       ClassExamFeature.objects.filter(exam=exam).delete()
       
       # 按班级分组统计
       class_stats = scores.values(
           'student__current_class',
           'student__current_school'
       ).annotate(
           class_id=F('student__current_class'),
           school_id=F('student__current_school'),
           avg_score=Avg('raw_score'),
           std_dev=StdDev('raw_score'),
           student_count=Count('student', distinct=True)
       )
       
       # 按学校分组，计算校内排名
       schools_classes = {}
       for class_stat in class_stats:
           school_id = class_stat['school_id']
           if school_id not in schools_classes:
               schools_classes[school_id] = []
           schools_classes[school_id].append(class_stat)
       
       class_features = {}
       for school_id, classes in schools_classes.items():
           sorted_classes = sorted(classes, key=lambda x: x['avg_score'], reverse=True)
           
           for rank, class_stat in enumerate(sorted_classes, 1):
               class_id = class_stat['class_id']
               school_percentile = ((len(sorted_classes) - rank) / len(sorted_classes)) * 100
               
               # 获取关联的学校特征
               school_feature = school_features.get(school_id)
               if not school_feature:
                   continue
               
               class_feature = ClassExamFeature.objects.create(
                   exam=exam,
                   school_feature=school_feature,
                   class_obj_id=class_id,
                   avg_score=class_stat['avg_score'],
                   std_dev=class_stat['std_dev'] or 0,
                   school_rank=rank,
                   school_percentile=school_percentile,
                   student_count=class_stat['student_count']
               )
               
               class_features[class_id] = class_feature
       
       return class_features
   ```

```javascript
function generateLayeredExamFeatures(examId) {
    // 显示状态面板
    featureStatus.style.display = 'block';
    statusMessage.innerHTML = '<span class="loading"></span> 正在生成考试分层特征...';
    progressFill.style.width = '10%';
    
    // 请求生成特征
    fetch('/api/admin/generate-exam-features/', {
        method: 'POST',
        headers: {
            'Content-Type': 'application/json',
            'X-CSRFToken': getCsrfToken()
        },
        body: JSON.stringify({
            exam_id: examId,
            regenerate: true
        })
    })
    .then(response => response.json())
    .then(data => {
        if (data.status === 'processing') {
            // 启动状态轮询
            progressFill.style.width = '30%';
            pollFeatureStatus(examId);
        } else if (data.exists) {
            statusMessage.innerHTML = '✓ 该考试已存在分层特征数据';
            progressFill.style.width = '100%';
        } else {
            statusMessage.innerHTML = '✗ 特征生成请求失败: ' + (data.error || '未知错误');
        }
    })
    .catch(error => {
        statusMessage.innerHTML = '✗ 请求出错: ' + error.message;
    });
}
```

```javascript
function pollFeatureStatus(examId) {
    let pollCount = 0;
    const maxPolls = 30; // 最多轮询30次
    
    const poll = function() {
        pollCount++;
        
        if (pollCount > maxPolls) {
            statusMessage.innerHTML = '⚠️ 特征计算超时，请检查后台日志';
            return;
        }
        
        fetch(`/api/admin/exam-feature-status/${examId}/`)
            .then(response => response.json())
            .then(data => {
                if (data.status === 'completed') {
                    // 特征计算完成
                    progressFill.style.width = '100%';
                    
                    // 显示统计信息
                    let message = '✓ 分层特征计算完成:<br>';
                    message += `- 区级特征: ${data.data.district_feature ? '已生成' : '未生成'}<br>`;
                    message += `- 学校特征: ${data.data.school_features}个<br>`;
                    message += `- 班级特征: ${data.data.class_features}个<br>`;
                    message += `- 学生特征: ${data.data.student_features}个`;
                    
                    statusMessage.innerHTML = message;
                } else if (data.status === 'error') {
                    // 特征计算错误
                    statusMessage.innerHTML = '✗ 特征计算错误: ' + data.message;
```

```python
def _generate_district_features(self, exam, scores):
    """生成区级特征"""
    # 删除已存在的特征
    DistrictExamFeature.objects.filter(exam=exam).delete()
    
    # 计算基础统计指标
    stats = scores.aggregate(
        avg=Avg('raw_score'),
        std=StdDev('raw_score'),
        min=Min('raw_score'),
        max=Max('raw_score'),
        count=Count('id')
    )
    
    # 计算各学科平均分
    subject_stats = scores.values('subject').annotate(
        subject_id=F('subject'),
        avg=Avg('raw_score')
    )
    
    subject_avgs = {str(stat['subject_id']): stat['avg'] for stat in subject_stats}
    
    # 计算参与学校、班级数量
    school_count = scores.values('student__current_school').distinct().count()
    class_count = scores.values('student__current_class').distinct().count()
    
    # 创建区级特征
    district_feature = DistrictExamFeature.objects.create(
        exam=exam,
        avg_score=stats['avg'] or 0,
        std_dev=stats['std'] or 0,
        total_students=stats['count'],
        max_score=stats['max'] or 0,
        min_score=stats['min'] or 0,
        subject_avg_scores=subject_avgs,
        school_count=school_count,
        class_count=class_count,
        student_count=stats['count']
    )
    
    logger.info(f"已生成考试 {exam.exam_id} 的区级特征")
```

```python
def _generate_school_features(self, exam, scores, district_feature):

    # 删除已存在的特征
    SchoolExamFeature.objects.filter(exam=exam).delete()
    
    # 按学校分组统计
    school_stats = scores.values('student__current_school').annotate(
        school_id=F('student__current_school'),
        avg_score=Avg('raw_score'),
        std_dev=StdDev('raw_score'),
        class_count=Count('student__current_class', distinct=True),
        top_score=Max('raw_score')
    ).order_by('-avg_score')
    
    # 创建学校特征
    school_features = {}
    
    for rank, school_stat in enumerate(school_stats, 1):
        school_id = school_stat['school_id']
        if not school_id:
            continue
            
        # 计算百分位 (0-100)
        district_percentile = ((len(school_stats) - rank) / len(school_stats)) * 100 if len(school_stats) > 0 else 0
        
        school_feature = SchoolExamFeature.objects.create(
            exam=exam,
            district_feature=district_feature,
            school_id=school_id,
            avg_score=school_stat['avg_score'] or 0,
            std_dev=school_stat['std_dev'] or 0,
            district_rank=rank,
            district_percentile=district_percentile,
            class_count=school_stat['class_count'],
            top_student_score=school_stat['top_score'] or 0
        )
        
        school_features[school_id] = school_feature
    
    logger.info(f"已生成考试 {exam.exam_id} 的 {len(school_features)} 个学校特征")
```

```python
def _generate_class_features(self, exam, scores, school_features):
    """生成班级特征"""
    # 删除已存在的特征
    ClassExamFeature.objects.filter(exam=exam).delete()
    
    # 按班级分组统计
    class_stats = scores.values(
        'student__current_class',
        'student__current_school'
    ).annotate(
        class_id=F('student__current_class'),
        school_id=F('student__current_school'),
        avg_score=Avg('raw_score'),
        std_dev=StdDev('raw_score'),
        student_count=Count('student', distinct=True)
    )
    
    # 按学校分组，计算班级在学校内的排名
    class_features = {}
    
    # 按学校ID分组班级统计
    schools_classes = {}
    for class_stat in class_stats:
        school_id = class_stat['school_id']
        if school_id not in schools_classes:
            schools_classes[school_id] = []
        schools_classes[school_id].append(class_stat)
    
    # 计算每个学校内班级的排名
    for school_id, classes in schools_classes.items():
        # 按平均分排序
        sorted_classes = sorted(classes, key=lambda x: x['avg_score'], reverse=True)
        
        for rank, class_stat in enumerate(sorted_classes, 1):
            class_id = class_stat['class_id']
            if not class_id:
                continue
                
            school_percentile = ((len(sorted_classes) - rank) / len(sorted_classes)) * 100 if len(sorted_classes) > 0 else 0
            
            # 获取关联的学校特征
            school_feature = school_features.get(school_id)
            if not school_feature:
                continue
            
            # 创建班级特征
            class_feature = ClassExamFeature.objects.create(
                exam=exam,
                school_feature=school_feature,
                class_obj_id=class_id,
                avg_score=class_stat['avg_score'] or 0,
                std_dev=class_stat['std_dev'] or 0,
                school_rank=rank,
                school_percentile=school_percentile,
                student_count=class_stat['student_count']
            )
            
            class_features[class_id] = class_feature
    
    logger.info(f"已生成考试 {exam.exam_id} 的 {len(class_features)} 个班级特征")
```
